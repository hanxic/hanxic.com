\newcommand{\CVEducationEntry}[1]{%
  \CVIfSet{education.entries.#1.cv_note}{%
    \CVSchool[\CVData{education.entries.#1.cv_note}]
      {\CVData{education.entries.#1.school}}
      {\CVData{education.entries.#1.cv_location}}%
  }{%
    \CVSchool
      {\CVData{education.entries.#1.school}}
      {\CVData{education.entries.#1.cv_location}}%
  }\\
  \foreach \degree in {1,...,\CVLen{education.entries.#1.degrees}}{%
    \textbf{\CVData{education.entries.#1.degrees.\degree.title}}%
    \CVIfSet{education.entries.#1.degrees.\degree.focus}{, \CVData{education.entries.#1.degrees.\degree.focus}}{}%
    \CVIfSet{education.entries.#1.degrees.\degree.cv_date}{%
      \hfill \CVData{education.entries.#1.degrees.\degree.cv_date}%
    }{%
      \ifnum\degree=1\relax\hfill \CVData{education.entries.#1.cv_years}\fi
    }\par
  }%
  \CVIfSet{education.entries.#1.gpa}{%
    GPA: \CVData{education.entries.#1.gpa}%
    \CVIfSet{education.entries.#1.distinction}{, \emph{\CVData{education.entries.#1.distinction}}}{}%
    \par
  }{}%
  \CVIfSet{education.entries.#1.advisors.@len}{\CVAdvisors{#1}\par}{}%
}

\usepackage[backend=biber,maxnames=99,style=authoryear,sorting=ydnt,defernumbers=true]{biblatex}
% ydnt breaks same-year ties on name/title, which put CoqPL (Jan) ahead of NFM
% (Jun). Sort on month too; entries without one fall to the end of their year.
\DeclareSortingTemplate{ymdnt}{
  \sort{\field{presort}}
  \sort[final]{\field{sortkey}}
  \sort[direction=descending]{\field{sortyear}\field{year}\literal{9999}}
  \sort[direction=descending]{\field[padside=left,padwidth=2,padchar=0]{month}\literal{00}}
  \sort{\field{sortname}\field{author}\field{editor}\field{sorttitle}\field{title}}
  \sort{\field{sorttitle}\field{title}}
}
\ExecuteBibliographyOptions{sorting=ymdnt}
% `award` and `talk` are nonstandard and not in biblatex's data model, so they
% are mapped onto scratch fields before they can be printed.

  \maps[datatype=bibtex]{
    \map{
      \step[fieldsource=award,fieldtarget=userb]
      \step[fieldsource=talk,fieldtarget=userc]
    }
  }
}

\newcount\SocietyItemCount
\newcommand{\HonorSocieties}{%
  \HonorCount{society}%
  \ifnum\HonorGroupCount>0
    \textbf{Honor societies}: %
    \global\SocietyItemCount=0
    \foreach \entry in {1,...,\CVLen{honors.entries}}{%
      \ifnum\pdfstrcmp{\CVRawData{honors.entries.\entry.group}}{society}=0
        \HonorIfShown{\entry}{%
          \ifnum\SocietyItemCount>0 , \fi

      \fi
    }%
    \par
  \fi

\newcommand{\ServiceLine}[1]{%

    \textbf{#1}: %
    \global\ServiceItemCount=0
    \foreach \entry in {1,...,\CVLen{service.entries}}{%
      \ifnum\pdfstrcmp{\CVRawData{service.entries.\entry.group}}{#1}=0
        \ifnum\ServiceItemCount>0 , \fi

          \CVData{service.entries.\entry.cv_label}%
        }{%
          \CVData{service.entries.\entry.organization}%

        \global\advance\ServiceItemCount by 1
      \fi
    }%
    \par
  \fi
