% resume_years overrides cv_years for the resume alone: the CV wants Penn's
% graduation date, the resume wants the span, and the CV must not change.
\newcommand{\ResumeEntryYears}[1]{%
  \CVIfSet{education.entries.#1.resume_years}{%
    \CVData{education.entries.#1.resume_years}%
  }{\CVData{education.entries.#1.cv_years}}%
}
% Place and time ride together on the school line so each institution carries
% exactly one date, instead of a date riding on whichever degree row is first.
\newcommand{\ResumeSchoolRight}[1]{%
  \CVData{education.entries.#1.cv_location}\Sep\ResumeEntryYears{#1}%
}
% resume_note likewise overrides cv_note. \ResumeSchool tests its optional
% argument with \detokenize, so the branch has to be chosen here rather than by
% handing it a macro that expands to nothing.
\newcommand{\ResumeSchoolLine}[1]{%
  \CVIfSet{education.entries.#1.resume_note}{%
    \ResumeSchool[\CVData{education.entries.#1.resume_note}]
      {\CVData{education.entries.#1.school}}{\ResumeSchoolRight{#1}}%
  }{%
    \CVIfSet{education.entries.#1.cv_note}{%
      \ResumeSchool[\CVData{education.entries.#1.cv_note}]
        {\CVData{education.entries.#1.school}}{\ResumeSchoolRight{#1}}%
    }{%
      \ResumeSchool
        {\CVData{education.entries.#1.school}}{\ResumeSchoolRight{#1}}%
    }%
  }%
}

\newcommand{\Sep}{\space\textbullet\space}

% Degrees print grouped by their resume_row (default 1) rather than strictly one
% per line, so an entry can fold short degrees onto a shared line when space is
% tight. Rows beyond the last populated one cost nothing.
\newcount\ResumeRowCount
\newcommand{\DegreeIfRow}[4]{%
  \CVIfSet{education.entries.#1.degrees.#2.resume_row}{%
    \ifnum\pdfstrcmp{\CVRawData{education.entries.#1.degrees.#2.resume_row}}{#3}=0
      #4%
    \fi
  }{%
    \ifnum#3=1\relax #4\fi
  }%
}
\newcommand{\ResumeDegreeRow}[2]{%
  \global\ResumeRowCount=0
  \foreach \degree in {1,...,\CVLen{education.entries.#1.degrees}}{%
    \DegreeIfRow{#1}{\degree}{#2}{%
      \ifnum\ResumeRowCount>0 ; \fi
      \textbf{\CVData{education.entries.#1.degrees.\degree.title}}%
      \CVIfSet{education.entries.#1.degrees.\degree.focus}{, \CVData{education.entries.#1.degrees.\degree.focus}}{}%
      % resume_school names the awarding school inline, where `focus` already
      % goes. Only worth setting where it tells the reader something the degree
      % does not already imply -- Wharton for the economics B.S.
      \CVIfSet{education.entries.#1.degrees.\degree.resume_school}{, \textit{\CVData{education.entries.#1.degrees.\degree.resume_school}}}{}%
      \global\advance\ResumeRowCount by 1
    }%
  }%
}
\newcommand{\ResumeEducationEntry}[1]{%
  \ResumeSchoolLine{#1}\\
  % No date on the degree rows: the school line above carries it, which leaves
  % every degree row with an empty right margin rather than only some of them.
  \ResumeDegreeRow{#1}{1}\par
  % The statement sits under the degree it describes rather than above the
  % section. Indented on both sides so it reads as a gloss on the Ph.D. line;
  % sized to wrap to exactly two lines at this measure.
  \CVIfSet{education.entries.#1.resume_statement}{%
    \CVIfSet{profile.statement.resume}{%
      {\leftskip=1.5em \rightskip=1.5em \noindent
        \CVData{profile.statement.resume}\par}%
    }{}%
  }{}%
  % Remaining rows: one line each, skipped silently when empty. GPA and awards
  % then start a line of their own so the degree list stays a degree list, but
  % they share that line with each other rather than taking one each. Entries
  % with none of the three (Cornell) emit no line at all.
  \ResumeDegreeRow{#1}{2}%
  \ifnum\ResumeRowCount>0 \par\fi
  \ResumeDegreeRow{#1}{3}%
  \ifnum\ResumeRowCount>0 \par\fi
  \global\ResumeRowCount=0

  \CVIfSet{education.entries.#1.gpa}{%
    \ifnum\ResumeRowCount>0 \Sep\fi
    GPA: \CVData{education.entries.#1.gpa}%
    \CVIfSet{education.entries.#1.distinction}{\Sep Honors: \emph{\CVData{education.entries.#1.distinction}}}{}%
    \global\advance\ResumeRowCount by 1
  }{}%
  \CVIfSet{education.entries.#1.resume_awards}{%
    \ifnum\ResumeRowCount>0 \Sep\fi
    Awards: \ResumeAwardsLine
    \global\advance\ResumeRowCount by 1
  }{}%
  \ifnum\ResumeRowCount>0 \par\fi
  \CVIfSet{education.entries.#1.advisors.@len}{\ResumeAdvisors{#1}\par}{}%
}

\newcount\ResumeAwardCount
\newcommand{\HonorIfShown}[2]{%
  \CVIfSet{honors.entries.#1.show.resume}{%
    \ifnum\pdfstrcmp{\CVRawData{honors.entries.#1.show.resume}}{false}=0
    \else
      #2%
    \fi
  }{#2}%
}
\newcommand{\ResumeAwardsLine}{%
  \global\ResumeAwardCount=0
  \foreach \entry in {1,...,\CVLen{honors.entries}}{%
    \ifnum\pdfstrcmp{\CVRawData{honors.entries.\entry.group}}{award}=0
      \HonorIfShown{\entry}{%
        % Commas, not bullets: this list is nested inside a bullet-separated
        % line, so reusing \Sep here would flatten the two levels together.
        \ifnum\ResumeAwardCount>0 , \fi
        \CVData{honors.entries.\entry.name}%
        \global\advance\ResumeAwardCount by 1
      }%
    \fi
  }%
}

\newcommand{\ResumeResearchEntry}[1]{%
  \begin{minipage}{\textwidth}
    \textbf{\CVData{research.entries.#1.role}}, \emph{\CVData{research.entries.#1.group}}%
    \CVIfSet{research.entries.#1.supervisors.@len}{, advised by \CVPrintList{research.entries.#1.supervisors}{ and }}{}%
    \hfill \CVData{research.entries.#1.period}

    \textbf{\CVData{research.entries.#1.institution}}%
    \CVIfSet{research.entries.#1.unit}{, \CVData{research.entries.#1.unit}}{}

    \begin{itemize}
      \foreach \bullet in {1,...,\CVLen{research.entries.#1.resume_bullets}}{%
        % A bullet is either a plain string or a {lead, note, text} map that
        % opens with a named artifact (the thesis titles). The lead is set plain:
        % bolding it competed with the entry headings for attention.
        \CVIfSet{research.entries.#1.resume_bullets.\bullet.lead}{%
          \item \CVData{research.entries.#1.resume_bullets.\bullet.lead}%
            \CVIfSet{research.entries.#1.resume_bullets.\bullet.note}{%
              \space(\CVData{research.entries.#1.resume_bullets.\bullet.note})%
            }{}%
            : \CVData{research.entries.#1.resume_bullets.\bullet.text}
        }{%
          \item \CVData{research.entries.#1.resume_bullets.\bullet}
        }%
      }%
    \end{itemize}
  \end{minipage}\par
}
