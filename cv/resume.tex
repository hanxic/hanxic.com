\newcommand{\CVPrintList}[2]{%
  \foreach \cvitem in {1,...,\CVLen{#1}}{%
    \CVData{#1.\cvitem}%
    \ifnum\cvitem<\CVLen{#1}\relax#2\fi
  }%
}

\newcommand{\ResumeHeader}{%
  {\centering
    {\LARGE \textbf{\CVData{profile.name.full}}}\\

    \CVLink{website}{website_label}%
    \CVLink{github}{github_label}%

\newcommand{\ResumeSchool}[3][]{%
  \textbf{#2}%
  \if\relax\detokenize{#1}\relax
  \else
    , \textit{#1}%
  \fi%
  \hfill #3%
}

\newcommand{\ResumeAdvisors}[1]{%
  Advisors:\space
  \foreach \advisor in {1,...,\CVLen{education.entries.#1.advisors}}{%
    \CVData{education.entries.#1.advisors.\advisor.title}~\CVData{education.entries.#1.advisors.\advisor.name}%
    \ifnum\advisor<\CVLen{education.entries.#1.advisors}\relax, \fi
  }%
}

\newcommand{\ResumeEducationEntry}[1]{%
  \CVIfSet{education.entries.#1.cv_note}{%
    \ResumeSchool[\CVData{education.entries.#1.cv_note}]
      {\CVData{education.entries.#1.school}}
      {\CVData{education.entries.#1.cv_location}}%
  }{%
    \ResumeSchool
      {\CVData{education.entries.#1.school}}
      {\CVData{education.entries.#1.cv_location}}%
  }\\
  \foreach \degree in {1,...,\CVLen{education.entries.#1.degrees}}{%
    \textbf{\CVData{education.entries.#1.degrees.\degree.title}}%
    \CVIfSet{education.entries.#1.degrees.\degree.focus}{, \CVData{education.entries.#1.degrees.\degree.focus}}{}%
    \CVIfSet{education.entries.#1.degrees.\degree.cv_date}{%
      \hfill \CVData{education.entries.#1.degrees.\degree.cv_date}%
    }{%
      \ifnum\degree=1\relax\hfill \CVData{education.entries.#1.cv_years}\fi
    }\par
  }%
  \CVIfSet{education.entries.#1.gpa}{%
    GPA: \CVData{education.entries.#1.gpa}%
    \CVIfSet{education.entries.#1.distinction}{, \emph{\CVData{education.entries.#1.distinction}}}{}%
    \par
  }{}%
  \CVIfSet{education.entries.#1.advisors.@len}{\ResumeAdvisors{#1}\par}{}%
}

\newcount\ResumeAwardCount
\newcommand{\HonorIfShown}[2]{%
  \CVIfSet{honors.entries.#1.show.cv}{%
    \ifnum\pdfstrcmp{\CVRawData{honors.entries.#1.show.cv}}{false}=0
    \else
      #2%
    \fi
  }{#2}%
}
\newcommand{\ResumeAwardsLine}{%
  \global\ResumeAwardCount=0
  \foreach \entry in {1,...,\CVLen{honors.entries}}{%
    \ifnum\pdfstrcmp{\CVRawData{honors.entries.\entry.group}}{award}=0
      \HonorIfShown{\entry}{%
        \ifnum\ResumeAwardCount>0 \space\textbullet\space \fi
        \CVData{honors.entries.\entry.name}%
        \global\advance\ResumeAwardCount by 1
      }%
    \fi
  }%
}

\newcommand{\ResumeResearchEntry}[1]{%
  \begin{minipage}{\textwidth}
    \textbf{\CVData{research.entries.#1.role}}, \emph{\CVData{research.entries.#1.group}}%
    \CVIfSet{research.entries.#1.supervisors.@len}{, advised by \CVPrintList{research.entries.#1.supervisors}{ and }}{}%
    \hfill \CVData{research.entries.#1.period}

    \textbf{\CVData{research.entries.#1.institution}}%
    \CVIfSet{research.entries.#1.unit}{, \CVData{research.entries.#1.unit}}{}

    \begin{itemize}
      \foreach \bullet in {1,...,\CVLen{research.entries.#1.resume_bullets}}{%
        \item \CVData{research.entries.#1.resume_bullets.\bullet}
      }%
    \end{itemize}
  \end{minipage}\par
}

\newcommand{\ResumeIfShown}[3]{%
  \CVIfSet{#1.entries.#2.show.resume}{%
    \ifnum\pdfstrcmp{\CVRawData{#1.entries.#2.show.resume}}{false}=0
    \else
      #3%
    \fi
  }{#3}%
}

\newcommand{\ResearchExperience}{%

    }%
    \vspace{-0.3em}
  }%
}

\newcommand{\ResumeProjectBullets}[1]{%
  \CVIfSet{projects.entries.#1.resume_bullets.@len}{%
    \foreach \bullet in {1,...,\CVLen{projects.entries.#1.resume_bullets}}{%
      \item \CVData{projects.entries.#1.resume_bullets.\bullet}
    }%
  }{%
    \foreach \bullet in {1,...,\CVLen{projects.entries.#1.bullets}}{%
      \item \CVData{projects.entries.#1.bullets.\bullet}
    }%
  }%
}

\newcommand{\ResumeProjectEntry}[1]{%
  \begin{minipage}{\textwidth}
    \textbf{\CVData{projects.entries.#1.title}} -- \emph{\CVData{projects.entries.#1.technologies.@joined}}%
    \hfill \CVData{projects.entries.#1.period}

    \textbf{\CVData{projects.entries.#1.institution}}%
    \CVIfSet{projects.entries.#1.course}{, \emph{\CVData{projects.entries.#1.course}}}{}

    \begin{itemize}
      \ResumeProjectBullets{#1}
    \end{itemize}
  \end{minipage}\par
}
